## 第1步：创建 ReviewRequest DTO

### 创建文件

`backend/src/main/java/com/buddy/dto/request/ReviewRequest.java`

### 写入代码

```java
package com.buddy.dto.request;

import lombok.Data;

import javax.validation.constraints.Max;
import javax.validation.constraints.Min;
import javax.validation.constraints.NotNull;
import javax.validation.constraints.Size;

/**
 * 评价请求
 */
@Data
public class ReviewRequest {

    /** 帖子ID */
    @NotNull
    private Long postId;

    /** 被评价用户ID */
    @NotNull
    private Long targetUserId;

    /** 评分 1-5星 */
    @NotNull
    @Min(1)
    @Max(5)
    private Integer rating;

    /** 标签：靠谱/爽约/迟到 */
    private String tag;

    /** 评价内容 */
    @Size(max = 500)
    private String content;
}
```

### 解释

- `@NotNull`：postId 和 targetUserId 不能为空，否则请求直接被拒绝
- `@Min(1) @Max(5)`：评分只能在 1-5 之间，超出范围会报参数错误
- `@Size(max = 500)`：评价内容最多 500 字
- `tag` 字段是可选的，用户可以选择"靠谱"、"爽约"、"迟到"等标签

## 第3步：创建 ReviewServiceImpl 实现类

### 创建文件

`backend/src/main/java/com/buddy/service/impl/ReviewServiceImpl.java`

### 写入代码

```java
package com.buddy.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.buddy.common.constant.ApplyStatus;
import com.buddy.common.constant.CreditConstant;
import com.buddy.common.constant.PostStatus;
import com.buddy.common.exception.BusinessException;
import com.buddy.common.result.ResultCode;
import com.buddy.dto.request.ReviewRequest;
import com.buddy.entity.Post;
import com.buddy.entity.PostApply;
import com.buddy.entity.Review;
import com.buddy.entity.User;
import com.buddy.mapper.PostApplyMapper;
import com.buddy.mapper.PostMapper;
import com.buddy.mapper.ReviewMapper;
import com.buddy.mapper.UserMapper;
import com.buddy.service.ReviewService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.time.temporal.ChronoUnit;
import java.util.List;

/**
 * 评价服务实现类
 */
@Service
public class ReviewServiceImpl implements ReviewService {

    /** 评价窗口期（天） */
    private static final int REVIEW_WINDOW_DAYS = 7;

    @Autowired
    private ReviewMapper reviewMapper;

    @Autowired
    private PostMapper postMapper;

    @Autowired
    private PostApplyMapper postApplyMapper;

    @Autowired
    private UserMapper userMapper;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void submitReview(Long reviewerId, ReviewRequest request) {
        // 1. 校验帖子状态为FINISHED
        Post post = postMapper.selectById(request.getPostId());
        if (post == null) {
            throw new BusinessException(ResultCode.NOT_FOUND, "帖子不存在");
        }
        if (!PostStatus.FINISHED.equals(post.getStatus())) {
            throw new BusinessException("只能评价已结束的活动");
        }

        // 2. 校验评价人是否参与了该活动（是发布者或已出勤的报名者）
        boolean isParticipant = post.getPublisherId().equals(reviewerId);
        if (!isParticipant) {
            Long attendedCount = postApplyMapper.selectCount(
                    new LambdaQueryWrapper<PostApply>()
                            .eq(PostApply::getPostId, request.getPostId())
                            .eq(PostApply::getApplicantId, reviewerId)
                            .eq(PostApply::getStatus, ApplyStatus.ATTENDED)
            );
            if (attendedCount == 0) {
                throw new BusinessException("您未参加该活动，无法评价");
            }
        }

        // 3. 校验被评价用户也参与了该活动
        boolean targetIsParticipant = post.getPublisherId().equals(request.getTargetUserId());
        if (!targetIsParticipant) {
            Long targetAttendedCount = postApplyMapper.selectCount(
                    new LambdaQueryWrapper<PostApply>()
                            .eq(PostApply::getPostId, request.getPostId())
                            .eq(PostApply::getApplicantId, request.getTargetUserId())
                            .eq(PostApply::getStatus, ApplyStatus.ATTENDED)
            );
            if (targetAttendedCount == 0) {
                throw new BusinessException("被评价用户未参加该活动");
            }
        }

        // 4. 校验不能评价自己
        if (reviewerId.equals(request.getTargetUserId())) {
            throw new BusinessException("不能评价自己");
        }

        // 5. 校验未重复评价
        Long reviewCount = reviewMapper.selectCount(
                new LambdaQueryWrapper<Review>()
                        .eq(Review::getPostId, request.getPostId())
                        .eq(Review::getReviewerId, reviewerId)
                        .eq(Review::getTargetUserId, request.getTargetUserId())
        );
        if (reviewCount > 0) {
            throw new BusinessException(ResultCode.REVIEW_DUPLICATE);
        }

        // 6. 校验7天评价窗口
        if (post.getUpdatedTime() != null) {
            long daysSinceFinished = ChronoUnit.DAYS.between(post.getUpdatedTime(), LocalDateTime.now());
            if (daysSinceFinished > REVIEW_WINDOW_DAYS) {
                throw new BusinessException(ResultCode.REVIEW_WINDOW_EXPIRED);
            }
        }

        // 7. 插入评价记录
        Review review = new Review();
        review.setPostId(request.getPostId());
        review.setReviewerId(reviewerId);
        review.setTargetUserId(request.getTargetUserId());
        review.setRating(request.getRating());
        review.setTag(request.getTag());
        review.setContent(request.getContent());
        reviewMapper.insert(review);

        // 8. 更新被评价者信誉分
        updateTargetCredit(request.getTargetUserId(), request.getRating());
    }

    @Override
    public List<Review> getPostReviews(Long postId) {
        return reviewMapper.selectList(
                new LambdaQueryWrapper<Review>()
                        .eq(Review::getPostId, postId)
                        .orderByDesc(Review::getCreatedTime)
        );
    }

    @Override
    public List<Review> getUserReviews(Long targetUserId) {
        return reviewMapper.selectList(
                new LambdaQueryWrapper<Review>()
                        .eq(Review::getTargetUserId, targetUserId)
                        .orderByDesc(Review::getCreatedTime)
        );
    }

    /**
     * 更新被评价者信誉分
     * 规则：1星-4分、5星+2分、中间按比例
     * 1星 → -4分
     * 2星 → -2分
     * 3星 → 0分
     * 4星 → +1分
     * 5星 → +2分
     */
    private void updateTargetCredit(Long targetUserId, Integer rating) {
        User targetUser = userMapper.selectById(targetUserId);
        if (targetUser == null) {
            return;
        }

        int creditChange = 0;
        switch (rating) {
            case 1:
                creditChange = -4;
                break;
            case 2:
                creditChange = -2;
                break;
            case 3:
                creditChange = 0;
                break;
            case 4:
                creditChange = 1;
                break;
            case 5:
                creditChange = 2;
                break;
            default:
                break;
        }

        int newScore = targetUser.getCreditScore() + creditChange;
        // 限制在0-100范围内
        newScore = Math.max(CreditConstant.MIN_SCORE, Math.min(CreditConstant.MAX_SCORE, newScore));
        targetUser.setCreditScore(newScore);
        userMapper.updateById(targetUser);
    }
}
```

### 解释

**submitReview() 方法的核心逻辑：**

1. **校验帖子状态**：只有 `FINISHED`（已结束）的帖子才能评价，防止活动还没结束就有人乱评价
2. **校验评价人资格**：评价人必须是帖子的发布者，或者是状态为 `ATTENDED`（已出勤）的报名者
3. **校验被评价人资格**：被评价人也必须是活动的参与者
4. **不能评价自己**：防止刷分
5. **不能重复评价**：同一个帖子中，A对B只能评价一次（数据库有唯一索引保障）
6. **7天评价窗口**：活动结束后7天内可以评价，超期不可评价
7. **插入评价记录**：通过 MyBatis-Plus 的 `insert` 方法保存
8. **更新信誉分**：根据评分动态调整被评价者的信誉分

**信誉分计算规则：**

| 评分 | 信誉分变化 |
|------|-----------|
| 1星 | -4分 |
| 2星 | -2分 |
| 3星 | 0分 |
| 4星 | +1分 |
| 5星 | +2分 |

这个规则体现了"差评重罚、好评轻奖"的设计思路，鼓励用户认真对待每一次活动。

## 第4步：创建 ReviewController

### 创建文件

`backend/src/main/java/com/buddy/controller/user/ReviewController.java`

### 写入代码

```java
package com.buddy.controller.user;

import com.buddy.common.context.UserContext;
import com.buddy.common.result.R;
import com.buddy.dto.request.ReviewRequest;
import com.buddy.entity.Review;
import com.buddy.service.ReviewService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import javax.validation.Valid;
import java.util.List;

/**
 * 评价控制器
 */
@RestController
@RequestMapping("/api/review")
public class ReviewController {

    @Autowired
    private ReviewService reviewService;

    /**
     * 提交评价
     */
    @PostMapping("/submit")
    public R<Void> submitReview(@RequestBody @Valid ReviewRequest request) {
        Long userId = UserContext.getUserId();
        reviewService.submitReview(userId, request);
        return R.ok();
    }

    /**
     * 帖子评价列表
     */
    @GetMapping("/list/{postId}")
    public R<List<Review>> getPostReviews(@PathVariable Long postId) {
        return R.ok(reviewService.getPostReviews(postId));
    }

    /**
     * 用户收到的评价
     */
    @GetMapping("/user/{userId}")
    public R<List<Review>> getUserReviews(@PathVariable Long userId) {
        return R.ok(reviewService.getUserReviews(userId));
    }
}
```

### 解释

- `@PostMapping("/submit")`：提交评价，需要登录（被JWT拦截器拦截）
- `@GetMapping("/list/{postId}")`：查看帖子的评价列表，公开接口（在WebMvcConfig中排除了JWT拦截）
- `@GetMapping("/user/{userId}")`：查看用户收到的评价，公开接口
- `@Valid`：触发参数校验，如果 ReviewRequest 中的校验注解不通过，会自动返回400错误

## 第5步：创建 FileController

### 创建文件

`backend/src/main/java/com/buddy/controller/user/FileController.java`

### 写入代码

```java
package com.buddy.controller.user;

import com.buddy.common.context.UserContext;
import com.buddy.common.exception.BusinessException;
import com.buddy.common.result.R;
import lombok.extern.slf4j.Slf4j;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.io.File;
import java.io.IOException;
import java.util.UUID;

/**
 * 文件上传控制器
 */
@Slf4j
@RestController
@RequestMapping("/api/file")
public class FileController {

    /** 上传目录 */
    private static final String UPLOAD_DIR = "uploads/avatar/";

    /** 允许的文件类型 */
    private static final String[] ALLOWED_TYPES = {"image/jpeg", "image/png", "image/jpg"};

    /** 最大文件大小：2MB */
    private static final long MAX_SIZE = 2 * 1024 * 1024;

    /**
     * 上传头像
     */
    @PostMapping("/upload")
    public R<String> uploadAvatar(@RequestParam("file") MultipartFile file) {
        Long userId = UserContext.getUserId();

        // 1. 校验文件是否为空
        if (file.isEmpty()) {
            throw new BusinessException("请选择要上传的文件");
        }

        // 2. 校验文件大小
        if (file.getSize() > MAX_SIZE) {
            throw new BusinessException("文件大小不能超过2MB");
        }

        // 3. 校验文件类型
        String contentType = file.getContentType();
        if (!isAllowedType(contentType)) {
            throw new BusinessException("仅支持jpg/png格式图片");
        }

        // 4. 生成文件名（UUID重命名，防止文件名冲突和路径穿越攻击）
        String originalFilename = file.getOriginalFilename();
        String extension = originalFilename != null && originalFilename.contains(".")
                ? originalFilename.substring(originalFilename.lastIndexOf("."))
                : ".jpg";
        String newFilename = UUID.randomUUID().toString().replace("-", "") + extension;

        // 5. 保存文件
        File uploadDir = new File(UPLOAD_DIR);
        if (!uploadDir.exists()) {
            uploadDir.mkdirs();
        }

        File dest = new File(uploadDir, newFilename);
        try {
            file.transferTo(dest);
        } catch (IOException e) {
            log.error("文件上传失败: userId={}, error={}", userId, e.getMessage());
            throw new BusinessException("文件上传失败，请重试");
        }

        // 6. 返回文件访问URL
        String url = "/uploads/avatar/" + newFilename;
        log.info("头像上传成功: userId={}, url={}", userId, url);
        return R.ok(url);
    }

    /**
     * 校验文件类型
     */
    private boolean isAllowedType(String contentType) {
        if (contentType == null) return false;
        for (String allowed : ALLOWED_TYPES) {
            if (allowed.equalsIgnoreCase(contentType)) return true;
        }
        return false;
    }
}
```

### 解释

**文件上传的安全要点：**

1. **校验文件大小**：限制2MB，防止大文件上传导致服务器磁盘被占满
2. **校验文件类型**：只允许 jpg/png，通过 `Content-Type` 判断，防止上传可执行文件
3. **UUID重命名**：不使用原始文件名，而是用UUID生成新文件名，原因：
   - 防止文件名冲突（不同用户上传同名文件）
   - 防止路径穿越攻击（恶意文件名如 `../../etc/passwd`）
4. **自动创建目录**：如果 `uploads/avatar/` 目录不存在，`mkdirs()` 会自动创建

**MultipartFile 常用方法：**

| 方法 | 作用 |
|------|------|
| `file.isEmpty()` | 判断文件是否为空 |
| `file.getSize()` | 获取文件大小（字节） |
| `file.getContentType()` | 获取文件MIME类型 |
| `file.getOriginalFilename()` | 获取原始文件名 |
| `file.transferTo(dest)` | 将文件保存到指定路径 |

## 第6步：配置静态资源映射

### 修改文件

`backend/src/main/java/com/buddy/config/WebMvcConfig.java`

### 写入代码

```java
package com.buddy.config;

import com.buddy.interceptor.JwtInterceptor;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.CorsRegistry;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.ResourceHandlerRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

/**
 * Web MVC 配置
 * 处理跨域、JWT 拦截器注册、静态资源映射等
 */
@Configuration
public class WebMvcConfig implements WebMvcConfigurer {

    @Autowired
    private JwtInterceptor jwtInterceptor;

    /**
     * 注册 JWT 拦截器，排除登录注册等公开接口
     */
    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        registry.addInterceptor(jwtInterceptor)
                .addPathPatterns("/api/**")
                .excludePathPatterns(
                        "/api/auth/**",
                        "/api/square/list",
                        "/api/post/detail/**",
                        "/api/post/comments/**",
                        "/api/review/list/**",
                        "/api/review/user/**",
                        "/api/admin/tag/list"
                );
    }

    /**
     * 配置 CORS 跨域
     */
    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/**")
                .allowedOriginPatterns("*")
                .allowedMethods("*")
                .allowedHeaders("*")
                .allowCredentials(true)
                .maxAge(3600);
    }

    /**
     * 配置静态资源映射（上传的图片访问）
     * 将 /uploads/** 的请求映射到本地 uploads 目录
     */
    @Override
    public void addResourceHandlers(ResourceHandlerRegistry registry) {
        registry.addResourceHandler("/uploads/**")
                .addResourceLocations("file:uploads/");
    }
}
```

### 解释

**静态资源映射的作用：**

上传的文件保存在 `uploads/avatar/` 目录下，但浏览器默认无法访问服务器上的本地文件。通过 `addResourceHandlers` 配置：

- `addResourceHandler("/uploads/**")`：拦截所有 `/uploads/` 开头的请求
- `addResourceLocations("file:uploads/")`：将这些请求映射到项目根目录下的 `uploads/` 文件夹

这样，当浏览器访问 `http://localhost:8080/uploads/avatar/xxx.jpg` 时，Spring 会去 `uploads/avatar/` 目录下找对应的文件并返回。

**JWT拦截器排除列表说明：**

| 排除路径 | 原因 |
|---------|------|
| `/api/auth/**` | 登录注册接口，不需要Token |
| `/api/square/list` | 广场列表，未登录也能看 |
| `/api/post/detail/**` | 帖子详情，未登录也能看 |
| `/api/post/comments/**` | 评论列表，未登录也能看 |
| `/api/review/list/**` | 评价列表，未登录也能看 |
| `/api/review/user/**` | 用户评价，未登录也能看 |
| `/api/admin/tag/list` | 标签列表，前端选择标签时使用 |
