```java
package com.buddy.service;

import com.baomidou.mybatisplus.core.metadata.IPage;
import com.buddy.dto.response.PostVO;

import java.util.Map;

/**
 * 管理后台服务接口

    /**
     * 获取用户列表（支持关键词搜索）

    IPage<UserVO> getUserList(int page, int size, String keyword);

     * 获取用户发布历史

     */

    /**
     * 获取用户评价记录
     */
    IPage<Map<String, Object>> getUserReviews(Long userId, int page, int size);

    /**
     * 封禁用户
     */
    void banUser(Long userId);

    /**
     * 解封用户
     */
    void unbanUser(Long userId);

    /**
     * 重置用户信誉分
     */
    void resetCredit(Long userId, Integer creditScore);

    /**
     * 获取帖子列表（管理后台，支持状态和分类筛选）
     */
    IPage<PostVO> getPostList(int page, int size, String status, String category, Integer isAuditPass);

     * 下架帖子
     */

     * 审核帖子

     */

```

### 解释

管理端服务接口分为三大块：

| 功能 | 方法 | 说明 |
|------|------|------|

| 帖子管理 | getPostList/takedownPost/auditPost/deletePost | 查看、下架、审核、删除 |
| 用户详情 | getUserPosts/getUserApplies/getUserReviews | 查看用户的发布/报名/评价记录 |

## 第2步：创建 AdminServiceImpl 实现类

### 创建文件

`backend/src/main/java/com/buddy/service/impl/AdminServiceImpl.java`

### 写入代码

```java
package com.buddy.service.impl;

import cn.hutool.core.util.StrUtil;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.buddy.common.constant.ApplyStatus;
import com.buddy.common.constant.PostStatus;
import com.buddy.common.exception.BusinessException;
import com.buddy.common.result.ResultCode;
import com.buddy.dto.response.PostVO;
import com.buddy.dto.response.UserVO;
import com.buddy.entity.*;
import com.buddy.mapper.*;
import com.buddy.service.AdminService;
import com.buddy.service.CreditService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * 管理后台服务实现类
 */
@Service
public class AdminServiceImpl implements AdminService {

    @Autowired
    private UserMapper userMapper;

    @Autowired
    private PostMapper postMapper;

    @Autowired
    private PostApplyMapper postApplyMapper;

    @Autowired
    private PostCommentMapper postCommentMapper;

    @Autowired
    private ReviewMapper reviewMapper;

    @Autowired
    private CreditService creditService;

    // ==================== 用户管理 ====================

    @Override
    public IPage<UserVO> getUserList(int page, int size, String keyword) {
        LambdaQueryWrapper<User> wrapper = new LambdaQueryWrapper<>();
        if (StrUtil.isNotBlank(keyword)) {
            wrapper.and(w -> w
                    .like(User::getNickname, keyword)
                    .or().like(User::getStudentNo, keyword)
                    .or().like(User::getPhone, keyword)
            );
        }
        wrapper.orderByDesc(User::getCreatedTime);

        IPage<User> userPage = userMapper.selectPage(new Page<>(page, size), wrapper);

        // 转换为UserVO
        IPage<UserVO> voPage = new Page<>(userPage.getCurrent(), userPage.getSize(), userPage.getTotal());
        voPage.setRecords(userPage.getRecords().stream()
                .map(this::convertToUserVO)
                .collect(Collectors.toList()));
        return voPage;
    }

    @Override
    public UserVO getUserDetail(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException(ResultCode.USER_NOT_FOUND);
        }
        return convertToUserVO(user);
    }

    @Override
    public IPage<PostVO> getUserPosts(Long userId, int page, int size) {
        LambdaQueryWrapper<Post> wrapper = new LambdaQueryWrapper<Post>()
                .eq(Post::getPublisherId, userId)
                .orderByDesc(Post::getCreatedTime);
        IPage<Post> postPage = postMapper.selectPage(new Page<>(page, size), wrapper);

        IPage<PostVO> voPage = new Page<>(postPage.getCurrent(), postPage.getSize(), postPage.getTotal());
        voPage.setRecords(postPage.getRecords().stream()
                .map(this::convertToPostVO)
                .collect(Collectors.toList()));
        return voPage;
    }

    @Override
    public IPage<Map<String, Object>> getUserApplies(Long userId, int page, int size) {
        LambdaQueryWrapper<PostApply> wrapper = new LambdaQueryWrapper<PostApply>()
                .eq(PostApply::getApplicantId, userId)
                .orderByDesc(PostApply::getApplyTime);
        IPage<PostApply> applyPage = postApplyMapper.selectPage(new Page<>(page, size), wrapper);

        IPage<Map<String, Object>> resultPage = new Page<>(applyPage.getCurrent(), applyPage.getSize(), applyPage.getTotal());
        resultPage.setRecords(applyPage.getRecords().stream().map(apply -> {
            Map<String, Object> map = new HashMap<>();
            map.put("id", apply.getId());
            map.put("postId", apply.getPostId());
            map.put("status", apply.getStatus());
            map.put("applyTime", apply.getApplyTime());
            map.put("handleTime", apply.getHandleTime());
            // 查询帖子标题
            Post post = postMapper.selectById(apply.getPostId());
            if (post != null) {
                map.put("postTitle", post.getTitle());
            }
            return map;
        }).collect(Collectors.toList()));
        return resultPage;
    }

    @Override
    public IPage<Map<String, Object>> getUserReviews(Long userId, int page, int size) {
        // 查询该用户收到的评价
        LambdaQueryWrapper<Review> wrapper = new LambdaQueryWrapper<Review>()
                .eq(Review::getTargetUserId, userId)
                .orderByDesc(Review::getCreatedTime);
        IPage<Review> reviewPage = reviewMapper.selectPage(new Page<>(page, size), wrapper);

        IPage<Map<String, Object>> resultPage = new Page<>(reviewPage.getCurrent(), reviewPage.getSize(), reviewPage.getTotal());
        resultPage.setRecords(reviewPage.getRecords().stream().map(review -> {
            Map<String, Object> map = new HashMap<>();
            map.put("id", review.getId());
            map.put("postId", review.getPostId());
            map.put("reviewerId", review.getReviewerId());
            map.put("rating", review.getRating());
            map.put("tag", review.getTag());
            map.put("content", review.getContent());
            map.put("createdTime", review.getCreatedTime());
            // 查询评价人昵称
            User reviewer = userMapper.selectById(review.getReviewerId());
            if (reviewer != null) {
                map.put("reviewerNickname", reviewer.getNickname());
            }
            // 查询帖子标题
            Post post = postMapper.selectById(review.getPostId());
            if (post != null) {
                map.put("postTitle", post.getTitle());
            }
            return map;
        }).collect(Collectors.toList()));
        return resultPage;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void banUser(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException(ResultCode.USER_NOT_FOUND);
        }
        user.setStatus(0);
        userMapper.updateById(user);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void unbanUser(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException(ResultCode.USER_NOT_FOUND);
        }
        user.setStatus(1);
        userMapper.updateById(user);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void resetCredit(Long userId, Integer creditScore) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException(ResultCode.USER_NOT_FOUND);
        }
        user.setCreditScore(creditScore);
        userMapper.updateById(user);
    }

    // ==================== 帖子管理 ====================

    @Override
    public IPage<PostVO> getPostList(int page, int size, String status, String category, Integer isAuditPass) {
        LambdaQueryWrapper<Post> wrapper = new LambdaQueryWrapper<>();

        if (StrUtil.isNotBlank(status)) {
            wrapper.eq(Post::getStatus, status);
        }
        if (StrUtil.isNotBlank(category)) {
            wrapper.eq(Post::getCategory, category);
        }
        if (isAuditPass != null) {
            wrapper.eq(Post::getIsAuditPass, isAuditPass);
        }
        wrapper.orderByDesc(Post::getCreatedTime);

        IPage<Post> postPage = postMapper.selectPage(new Page<>(page, size), wrapper);

        // 转换为PostVO
        IPage<PostVO> voPage = new Page<>(postPage.getCurrent(), postPage.getSize(), postPage.getTotal());
        voPage.setRecords(postPage.getRecords().stream()
                .map(this::convertToPostVO)
                .collect(Collectors.toList()));
        return voPage;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void takedownPost(Long postId) {
        Post post = postMapper.selectById(postId);
        if (post == null) {
            throw new BusinessException(ResultCode.NOT_FOUND, "帖子不存在");
        }
        // 下架即将帖子状态设为CANCELLED
        post.setStatus(PostStatus.CANCELLED);
        postMapper.updateById(post);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void auditPost(Long postId, boolean pass) {
        Post post = postMapper.selectById(postId);
        if (post == null) {
            throw new BusinessException(ResultCode.NOT_FOUND, "帖子不存在");
        }
        post.setIsAuditPass(pass ? 1 : 0);
        postMapper.updateById(post);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void deletePost(Long postId) {
        Post post = postMapper.selectById(postId);
        if (post == null) {
            throw new BusinessException(ResultCode.NOT_FOUND, "帖子不存在");
        }
        // 物理删除：先删关联数据，再删帖子
        postApplyMapper.delete(new LambdaQueryWrapper<PostApply>().eq(PostApply::getPostId, postId));
        postCommentMapper.delete(new LambdaQueryWrapper<PostComment>().eq(PostComment::getPostId, postId));
        reviewMapper.delete(new LambdaQueryWrapper<Review>().eq(Review::getPostId, postId));
        postMapper.deleteById(postId);
    }

    // ==================== 私有方法 ====================

    /**
     * 将User实体转换为UserVO
     */
    private UserVO convertToUserVO(User user) {
        UserVO vo = new UserVO();
        vo.setId(user.getId());
        vo.setStudentNo(user.getStudentNo());
        vo.setPhone(user.getPhone());
        vo.setNickname(user.getNickname());
        vo.setAvatar(user.getAvatar());
        vo.setSignature(user.getSignature());
        vo.setCollege(user.getCollege());
        vo.setGender(user.getGender());
        vo.setCreditScore(user.getCreditScore());
        vo.setCreditLevel(creditService.getCreditLevel(user.getCreditScore()));
        vo.setCreditLabel(creditService.getCreditLabel(user.getCreditScore()));
        vo.setStatus(user.getStatus());
        vo.setCreatedTime(user.getCreatedTime());
        return vo;
    }

    /**
     * 将Post实体转换为PostVO
     */
    private PostVO convertToPostVO(Post post) {
        PostVO vo = new PostVO();
        vo.setId(post.getId());
        vo.setTitle(post.getTitle());
        vo.setCategory(post.getCategory());
        vo.setTagIds(post.getTagIds());
        vo.setStartTime(post.getStartTime());
        vo.setEndTime(post.getEndTime());
        vo.setLocation(post.getLocation());
        vo.setMaxPeople(post.getMaxPeople());
        vo.setCurrentPeople(post.getCurrentPeople());
        vo.setStatus(post.getStatus());
        vo.setViewCount(post.getViewCount());
        vo.setPublisherId(post.getPublisherId());
        vo.setCreatedTime(post.getCreatedTime());

        // 查询发布者信息
        User publisher = userMapper.selectById(post.getPublisherId());
        if (publisher != null) {
            vo.setPublisherNickname(publisher.getNickname());
            vo.setPublisherAvatar(publisher.getAvatar());
            vo.setPublisherCreditLevel(creditService.getCreditLevel(publisher.getCreditScore()));
        }
        return vo;
    }
}
```

### 解释

**用户管理核心逻辑：**

- `banUser()`：将用户 `status` 设为0，被封禁的用户无法登录（登录时校验status）
- `unbanUser()`：将用户 `status` 设为1，恢复正常
- `resetCredit()`：管理员手动重置用户信誉分，用于处理误判或特殊情况

**帖子管理核心逻辑：**

- `getPostList()`：支持按状态、分类、审核状态筛选，`isAuditPass` 参数可以筛选待审核帖子
- `takedownPost()`：将帖子状态设为 `CANCELLED`，这是逻辑删除，帖子数据还在
- `auditPost()`：审核帖子，`pass=true` 设为1（通过），`pass=false` 设为0（拒绝）
- `deletePost()`：**物理删除**，先删关联的报名记录、评论、评价，再删帖子本身

**物理删除的顺序很重要：**

```
1. 删报名记录（post_apply）
2. 删评论（post_comment）
3. 删评价（review）
4. 删帖子（post）
```

必须先删子表数据再删主表，否则会因为外键约束报错。

```java
package com.buddy.controller.admin;

import com.baomidou.mybatisplus.core.metadata.IPage;
import com.buddy.common.annotation.RequireAdmin;
import com.buddy.common.result.R;
import com.buddy.dto.response.PostVO;
import com.buddy.dto.response.UserVO;
import com.buddy.service.AdminService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

/**
 * 管理端-用户管理控制器
 */
@RestController
@RequestMapping("/api/admin/user")
public class AdminUserController {

    @Autowired
    private AdminService adminService;

    /**
     * 用户列表
     */
    @RequireAdmin
    @GetMapping("/list")
    public R<IPage<UserVO>> getUserList(
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "10") int size,
            @RequestParam(required = false) String keyword) {
        return R.ok(adminService.getUserList(page, size, keyword));
    }

    /**
     * 用户详情
     */
    @RequireAdmin
    @GetMapping("/detail/{id}")
    public R<UserVO> getUserDetail(@PathVariable Long id) {
        return R.ok(adminService.getUserDetail(id));
    }

    /**
     * 用户发布历史
     */
    @RequireAdmin
    @GetMapping("/posts/{id}")
    public R<IPage<PostVO>> getUserPosts(
            @PathVariable Long id,
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "5") int size) {
        return R.ok(adminService.getUserPosts(id, page, size));
    }

    /**
     * 用户报名历史
     */
    @RequireAdmin
    @GetMapping("/applies/{id}")
    public R<IPage<Map<String, Object>>> getUserApplies(
            @PathVariable Long id,
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "5") int size) {
        return R.ok(adminService.getUserApplies(id, page, size));
    }

    /**
     * 用户评价记录
     */
    @RequireAdmin
    @GetMapping("/reviews/{id}")
    public R<IPage<Map<String, Object>>> getUserReviews(
            @PathVariable Long id,
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "5") int size) {
        return R.ok(adminService.getUserReviews(id, page, size));
    }

    /**
     * 封禁用户
     */
    @RequireAdmin
    @PostMapping("/ban/{id}")
    public R<Void> banUser(@PathVariable Long id) {
        adminService.banUser(id);
        return R.ok();
    }

    /**
     * 解封用户
     */
    @RequireAdmin
    @PostMapping("/unban/{id}")
    public R<Void> unbanUser(@PathVariable Long id) {
        adminService.unbanUser(id);
        return R.ok();
    }

    /**
     * 重置信誉分
     * body: {userId: 1, creditScore: 80}
     */
    @RequireAdmin
    @PutMapping("/credit")
    public R<Void> resetCredit(@RequestBody Map<String, Object> body) {
        Long userId = Long.valueOf(body.get("userId").toString());
        Integer creditScore = Integer.valueOf(body.get("creditScore").toString());
        adminService.resetCredit(userId, creditScore);
        return R.ok();
    }
}
```

### 解释

- 所有接口都加了 `@RequireAdmin` 注解，JWT拦截器会检查当前用户是否为管理员
- `resetCredit` 使用 `@RequestBody Map<String, Object>` 接收参数，因为参数简单不需要单独创建DTO
- 用户详情页可以查看该用户的发布历史、报名历史、评价记录，方便管理员判断是否封禁

```java
package com.buddy.controller.admin;

import com.baomidou.mybatisplus.core.metadata.IPage;
import com.buddy.common.annotation.RequireAdmin;
import com.buddy.common.result.R;
import com.buddy.dto.response.PostVO;
import com.buddy.service.AdminService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.Map;

/**
 * 管理端-帖子管理控制器
 */
@RestController
@RequestMapping("/api/admin/post")
public class AdminPostController {

    @Autowired
    private AdminService adminService;

    /**
     * 帖子列表（支持审核状态筛选）
     */
    @RequireAdmin
    @GetMapping("/list")
    public R<IPage<PostVO>> getPostList(
            @RequestParam(defaultValue = "1") int page,
            @RequestParam(defaultValue = "10") int size,
            @RequestParam(required = false) String status,
            @RequestParam(required = false) String category,
            @RequestParam(required = false) Integer isAuditPass) {
        return R.ok(adminService.getPostList(page, size, status, category, isAuditPass));
    }

    /**
     * 下架帖子
     */
    @RequireAdmin
    @PostMapping("/takedown/{id}")
    public R<Void> takedownPost(@PathVariable Long id) {
        adminService.takedownPost(id);
        return R.ok();
    }

    /**
     * 审核帖子
     * body: {pass: true}
     */
    @RequireAdmin
    @PostMapping("/audit/{id}")
    public R<Void> auditPost(@PathVariable Long id, @RequestBody Map<String, Object> body) {
        boolean pass = Boolean.parseBoolean(body.get("pass").toString());
        adminService.auditPost(id, pass);
        return R.ok();
    }

    /**
     * 物理删除帖子
     */
    @RequireAdmin
    @DeleteMapping("/delete/{id}")
    public R<Void> deletePost(@PathVariable Long id) {
        adminService.deletePost(id);
        return R.ok();
    }
}
```

### 解释

- `isAuditPass` 参数：0=待审核，1=已通过，null=全部。管理员可以筛选待审核的帖子
- `takedownPost`：逻辑删除，将帖子状态改为CANCELLED
- `auditPost`：审核帖子，传入 `{pass: true}` 通过审核，传入 `{pass: false}` 拒绝
- `deletePost`：物理删除，使用 `@DeleteMapping` 语义更清晰

## 第6步：创建 AdminTagController

### 创建文件

`backend/src/main/java/com/buddy/controller/admin/AdminTagController.java`

### 写入代码

```java
package com.buddy.controller.admin;

import com.buddy.common.annotation.RequireAdmin;
import com.buddy.common.result.R;
import com.buddy.entity.InterestTag;
import com.buddy.service.TagService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.List;

/**
 * 管理端-标签管理控制器
 */
@RestController
@RequestMapping("/api/admin/tag")
public class AdminTagController {

    @Autowired
    private TagService tagService;

    /**
     * 标签列表（管理端查看，包含禁用的）
     */
    @GetMapping("/list")
    public R<List<InterestTag>> getAllTags() {
        return R.ok(tagService.getAllTags());
    }

    /**
     * 获取所有分类名称列表
     */
    @GetMapping("/categories")
    public R<List<String>> getAllCategories() {
        return R.ok(tagService.getAllCategories());
    }

    /**
     * 添加标签
     */
    @RequireAdmin
    @PostMapping("/add")
    public R<InterestTag> addTag(@RequestBody InterestTag tag) {
        return R.ok(tagService.addTag(tag));
    }

    /**
     * 编辑标签
     */
    @RequireAdmin
    @PutMapping("/update")
    public R<InterestTag> updateTag(@RequestBody InterestTag tag) {
        return R.ok(tagService.updateTag(tag));
    }

    /**
     * 禁用标签
     */
    @RequireAdmin
    @PostMapping("/disable/{id}")
    public R<Void> disableTag(@PathVariable Long id) {
        tagService.disableTag(id);
        return R.ok();
    }
}
```

### 解释

- `/list` 和 `/categories` 没有加 `@RequireAdmin`，因为前端用户端也需要调用（选择标签时）
- `/add`、`/update`、`/disable` 加了 `@RequireAdmin`，只有管理员才能增删改标签
- 禁用标签是逻辑删除（status设为0），不是物理删除，因为可能有关联数据

### 解释

- 所有统计接口都加了 `@RequireAdmin`，只有管理员才能查看数据统计
